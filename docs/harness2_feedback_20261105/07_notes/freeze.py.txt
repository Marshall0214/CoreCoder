def read(path):
    return json.loads(path.read_text(encoding='utf-8'))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def copy(source, name):
    assert source.is_file() and not source.is_symlink(), source
    target = PACKET / name
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, target)

def validation(path, target):
    for name in ('report.json', 'patch.diff'):
        copy(path / name, target + '/' + name)
    for folder in ('static-baseline', 'static-candidate', 'public-logs'):
        for file in sorted((path / folder).iterdir()):
            if file.is_file() and file.suffix in ('.json', '.txt', '.py'):
                copy(file, target + '/' + folder + '/' + file.name)
